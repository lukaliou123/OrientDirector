## 🎯 前端集成建议

### 1. 获取可用模板 (JavaScript)

```javascript
// 获取可用的梗图模板
async function loadMemeTemplates() {
    try {
        const response = await fetch(`${API_BASE}/api/meme-templates`);
        const data = await response.json();
        
        if (data.success) {
            console.log('✅ 模板加载成功:', data.templates);
            displayTemplateButtons(data.templates);
        }
    } catch (error) {
        console.error('❌ 模板加载失败:', error);
    }
}

// 显示模板选择按钮
function displayTemplateButtons(templates) {
    const container = document.getElementById('templateButtonsContainer');
    
    templates.forEach(template => {
        const button = document.createElement('button');
        button.className = 'btn template-btn';
        button.innerHTML = `
            <span class="template-name">${template.name}</span>
            <span class="template-desc">${template.description}</span>
        `;
        button.onclick = () => selectTemplate(template.id, template.name);
        container.appendChild(button);
    });
}
```

### 2. 模板选择功能

```javascript
let selectedTemplate = null;

// 选择模板
function selectTemplate(templateId, templateName) {
    selectedTemplate = templateId;
    
    // 更新UI显示选中状态
    document.querySelectorAll('.template-btn').forEach(btn => {
        btn.classList.remove('selected');
    });
    event.target.classList.add('selected');
    
    // 显示选中的模板
    document.getElementById('selectedTemplate').textContent = templateName;
    
    // 可选：清空自定义prompt输入框
    document.getElementById('memePromptInput').value = '';
    
    console.log(`📋 已选择模板: ${templateName} (${templateId})`);
}

// 清除模板选择
function clearTemplate() {
    selectedTemplate = null;
    document.querySelectorAll('.template-btn').forEach(btn => {
        btn.classList.remove('selected');
    });
    document.getElementById('selectedTemplate').textContent = '无';
}
```

### 3. 生成梗图（修改现有函数）

```javascript
// 修改现有的生成梗图函数
async function generateMeme() {
    const formData = new FormData();
    
    // 基础参数
    formData.append('character_image', characterImageFile);
    if (compositionImageFile) {
        formData.append('composition_image', compositionImageFile);
    }
    formData.append('scene_elements', JSON.stringify(state.sceneElements));
    formData.append('historical_info', JSON.stringify(state.historicalInfo));
    
    // 提示词处理
    if (selectedTemplate) {
        // 使用模板时，meme_prompt可以为空或用作补充说明
        const customNotes = document.getElementById('memePromptInput').value || '';
        formData.append('meme_prompt', customNotes);
        formData.append('template_id', selectedTemplate);
        console.log(`🎯 使用模板生成: ${selectedTemplate}`);
    } else {
        // 自定义模式
        const customPrompt = document.getElementById('memePromptInput').value;
        if (!customPrompt) {
            alert('请输入梗图创意提示词或选择预设模板');
            return;
        }
        formData.append('meme_prompt', customPrompt);
        console.log('📝 使用自定义提示词生成');
    }
    
    try {
        showLoading(true);
        
        const response = await fetch(`${API_BASE}/api/generate-historical-meme`, {
            method: 'POST',
            body: formData
        });
        
        const data = await response.json();
        
        if (data.success) {
            displayGeneratedMeme(data.meme_url);
            console.log('🎉 梗图生成成功:', data);
        } else {
            alert('梗图生成失败');
        }
    } catch (error) {
        console.error('❌ 梗图生成错误:', error);
        alert('梗图生成失败，请重试');
    } finally {
        showLoading(false);
    }
}
```

## 🎯 工作流程

1. **页面加载时**: 调用`loadMemeTemplates()`加载模板按钮
2. **用户选择模板**: 点击模板按钮，自动填充预设prompt
3. **用户上传图片**: 人物素材（必需）+ 构图参考（可选）
4. **生成梗图**: 
   - 自动翻译中文场景元素为英文
   - 自动替换[year]、[location]等占位符
   - 多模态生成：文字 + 人物图 + 构图图
